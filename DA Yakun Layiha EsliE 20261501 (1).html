# Option 1: Convert to percentage format
ag <- ag %>%
  mutate(profit_margin_pct = profit_margin * 100)

# Then recreate categories with the new column
ag <- ag %>% 
  mutate(marja_category = case_when(
    profit_margin_pct >= 200 ~ ">=200",
    profit_margin_pct >= 100 & profit_margin_pct < 200 ~ "100-200",
    profit_margin_pct >= 0 & profit_margin_pct < 100 ~ "0-100",
    profit_margin_pct >= -100 & profit_margin_pct < 0 ~ "-100-0",
    profit_margin_pct >= -200 & profit_margin_pct < -100 ~ "-200--100",
    profit_margin_pct < -200 ~ "<-200",
    TRUE ~ "NA"
  ))

# Create summary by category
